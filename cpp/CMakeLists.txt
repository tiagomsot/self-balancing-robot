cmake_minimum_required(VERSION 3.16)
project(balancing_robot_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(ROBOT_CORE_SOURCES src/estimator.cpp src/controller.cpp src/behaviour.cpp
    src/safety.cpp src/motor_command.cpp src/my_routine.cpp)
add_library(robot_portable STATIC ${ROBOT_CORE_SOURCES})
set_target_properties(robot_portable PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(robot_portable PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_options(robot_portable PRIVATE -Wall -Wextra -Wpedantic)
add_library(robot_core SHARED targets/host/robot_c_api.cpp targets/mujoco/warp_host.cpp)
target_link_libraries(robot_core PRIVATE robot_portable)
if(PROJECT_IS_TOP_LEVEL OR CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    enable_testing()
    add_executable(robot_core_tests tests/core_tests.cpp)
    target_link_libraries(robot_core_tests PRIVATE robot_portable)
    add_test(NAME portable_core COMMAND robot_core_tests)
endif()
